RESTful接口设计与Axios封装案例
一、RESTful接口设计原则

GET：获取资源（如查询用户列表）
POST：创建资源（如新增用户）
PUT：更新完整资源（如修改用户全部信息）
PATCH：更新部分资源（如仅修改用户邮箱）
DELETE：删除资源（如移除用户）

200 OK：成功请求
201 Created：资源创建成功
400 Bad Request：客户端参数错误
401 Unauthorized：未认证
404 Not Found：资源不存在
500 Internal Server Error：服务端错误


二、Axios封装实现
封装目标：统一处理请求配置、错误拦截、身份认证和响应数据

// 创建axios实例
const service = axios.create({
  baseURL: 'https://api.example.com', // 基础路径
  timeout: 10000, // 超时时间
  headers: { 'Content-Type': 'application/json' }
});

// 请求拦截器：添加token
service.interceptors.request.use(
  config => {
    const token = localStorage.getItem('token');
    if (token) config.headers.Authorization = `Bearer ${token}`;
    return config;
  },
  error => Promise.reject(error)
);

// 响应拦截器：统一处理错误
service.interceptors.response.use(
  response => {
    // 提取核心数据（根据后端约定）
    return response.data; 
  },
  error => {
    // 根据HTTP状态码处理
    if (error.response) {
      switch (error.response.status) {
        case 401: 
          console.error('未授权，跳转登录页');
          break;
        case 404:
          console.error('资源不存在');
          break;
        default:
          console.error('服务端错误:', error.message);
      }
    }
    return Promise.reject(error);
  }
);

// 封装通用请求方法
export const http = {
  get: (url, params) => service.get(url, { params }),
  post: (url, data) => service.post(url, data),
  put: (url, data) => service.put(url, data),
  delete: url => service.delete(url)
};

// 用户模块接口调用
import { http } from './axios-service';

// 获取用户列表
const fetchUsers = async () => {
  try {
    const res = await http.get('/users');
    console.log('用户数据:', res);
  } catch (err) {
    console.error('请求失败:', err);
  }
};

// 创建新用户
const createUser = async (userData) => {
  await http.post('/users', userData); // 自动携带token
};

// 更新用户信息
const updateUser = async (id, data) => {
  await http.put(`/users/${id}`, data);
};

统一错误处理

增强安全性
自动注入身份凭证，避免遗漏
